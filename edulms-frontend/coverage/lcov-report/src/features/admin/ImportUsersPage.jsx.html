                                  <AlertTriangle className="w-3 h-3 text-red-400 mt-0.5 shrink-0" />
                                  {err}
                                </li>
                              ))}
                            </ul>
                          </td>
                        </tr>
                      ))}
                    </tbody>
                  </table>
                </div>
              )}
            </div>
          )}
        </div>
      )}
    </div>
  )
}
